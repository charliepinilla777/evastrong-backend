📋 VARIABLES DE ENTORNO REQUERIDAS
───────────────────────────────────────────────────────────────────────────────

PRODUCCIÓN (Render):

  MONGODB_URI = mongodb+srv://usuario:password@cluster.mongodb.net/evastrong
  JWT_SECRET = [generar con: node -e "...randomBytes(32).toString('hex')"]

DESARROLLO (Local):

  NODE_ENV = development
  MONGODB_URI = mongodb://localhost:27017/evastrong
  JWT_SECRET = [generar con: node -e "...randomBytes(32).toString('hex')"]
  FRONTEND_URL = http://localhost:3000
